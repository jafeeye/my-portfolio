#!/bin/bash

# =====================================================================
# Illumio PCE 26.2.10
# Rocky Linux Offline Dependency Downloader
#
# 功能：
#
#   1. 專門下載 Illumio PCE 26.2.10 所需 OS prerequisites
#   2. 支援 Rocky Linux：
#
#        9.0 ~ 9.8
#        10.0 ~ 10.2
#
#   3. 使用 Rocky 指定版本 repository
#   4. 使用 --resolve 自動下載 prerequisite 的 dependency
#   5. 使用 --alldeps，已安裝在下載主機上的 dependency 也會下載
#   6. 不加入 illumio-pce RPM
#   7. 不打 tar.gz
#   8. 不建立 repodata
#
#
# 使用：
#
#   ./pce-deps.sh 9.8
#
#   ./pce-deps.sh 10.2
#
# =====================================================================


set -Eeuo pipefail


# =====================================================================
# Illumio PCE 26.2.10 官方 OS prerequisites
# =====================================================================
#
# 注意：
#
# syslog 部分官方要求：
#
#     syslog-ng >= 3.23
#          OR
#     rsyslog
#
# Rocky Linux 預設就是 rsyslog，所以這裡選 rsyslog。
#
# =====================================================================

PCE_PACKAGES=(

    # -------------------------------------------------------------
    # 基本 shell / utilities
    # -------------------------------------------------------------

    bash
    bzip2
    chkconfig
    coreutils
    findutils
    gawk
    grep


    # -------------------------------------------------------------
    # Runtime / system libraries
    # -------------------------------------------------------------

    glibc
    zlib


    # -------------------------------------------------------------
    # SysV / service compatibility
    # -------------------------------------------------------------

    initscripts


    # -------------------------------------------------------------
    # Logging
    # -------------------------------------------------------------

    logrotate
    rsyslog


    # -------------------------------------------------------------
    # Network utilities
    # -------------------------------------------------------------

    net-tools


    # -------------------------------------------------------------
    # Process utilities
    #
    # Rocky / RHEL package 名稱實際是 procps-ng
    # Illumio 文件描述為 procps >= 3.2.0
    # -------------------------------------------------------------

    procps-ng


    # -------------------------------------------------------------
    # System utilities
    # -------------------------------------------------------------

    sed
    shadow-utils
    tar
    util-linux
    
    # ---------------------------
    # Others
    # ---------------------------
    chrony
    glibc-langpack-en
    
    
)


# =====================================================================
# 額外 shared libraries
#
# 這些是 PCE 在 EL 系統上常見的 runtime libraries，
# 並且你前面準備 PCE 環境時也已經使用。
#
# 若 OS 原本已有，--alldeps 仍會帶入完整 offline bundle。
# =====================================================================

PCE_SHARED_LIBS=(

    libxml2
    libuuid
    ncurses-libs

    openssl-libs

    libxcrypt

    libstdc++
    libgcc

    xz-libs
)


# =====================================================================
# 合併 Package List
# =====================================================================

PACKAGES=(

    "${PCE_PACKAGES[@]}"
    "${PCE_SHARED_LIBS[@]}"

)


# =====================================================================
# Settings
# =====================================================================

ARCH="x86_64"

OUTPUT_ROOT="/agent-download/illumio-pce-dependencies"


# =====================================================================
# Functions
# =====================================================================

info()
{
    echo
    echo "[INFO] $*"
}


warn()
{
    echo
    echo "[WARN] $*"
}


die()
{
    echo
    echo "[ERROR] $*" >&2
    exit 1
}


# =====================================================================
# Root Check
# =====================================================================

if [[ "${EUID}" -ne 0 ]]; then

    die "請使用 root 執行"

fi


# =====================================================================
# Argument Check
# =====================================================================

if [[ $# -ne 1 ]]; then

    echo
    echo "使用方式："
    echo
    echo "  $0 <Rocky Linux version>"
    echo
    echo "例如："
    echo
    echo "  $0 9.8"
    echo "  $0 10.2"
    echo

    exit 1

fi


VERSION="$1"


# =====================================================================
# Supported Versions
# =====================================================================

case "${VERSION}" in

    9.0|9.1|9.2|9.3|9.4|9.5|9.6|9.7|9.8|\
    10.0|10.1|10.2)

        ;;

    *)

        die "只支援 Rocky Linux 9.0 ~ 9.8、10.0 ~ 10.2"

        ;;

esac


# =====================================================================
# Major Version
# =====================================================================

MAJOR="${VERSION%%.*}"


# =====================================================================
# Check dnf
# =====================================================================

if ! command -v dnf >/dev/null 2>&1; then

    die "找不到 dnf"

fi


# =====================================================================
# Check dnf download
# =====================================================================

if ! dnf download --help >/dev/null 2>&1; then

    info "dnf download 不存在，嘗試安裝 dnf-plugins-core"

    dnf install -y dnf-plugins-core || true

fi


if ! dnf download --help >/dev/null 2>&1; then

    die "dnf download 無法使用"

fi


# =====================================================================
# Output Directory
# =====================================================================

OUTPUT_DIR="${OUTPUT_ROOT}/rocky-${VERSION}"

RPM_DIR="${OUTPUT_DIR}/packages"


mkdir -p "${RPM_DIR}"


# =====================================================================
# Rocky Repository
# =====================================================================
#
# 舊版本固定 Vault。
#
# 目前 current release 可以用 pub repository。
#
# 這裡：
#
#   10.2 -> current
#
# 其他 -> vault
#
# =====================================================================

case "${VERSION}" in

    10.2)

        BASE_URL="https://dl.rockylinux.org/pub/rocky/${VERSION}"
        ;;


    *)

        BASE_URL="https://dl.rockylinux.org/vault/rocky/${VERSION}"
        ;;

esac


# =====================================================================
# Repository URLs
# =====================================================================

BASEOS_URL="${BASE_URL}/BaseOS/${ARCH}/os/"

APPSTREAM_URL="${BASE_URL}/AppStream/${ARCH}/os/"

CRB_URL="${BASE_URL}/CRB/${ARCH}/os/"


# =====================================================================
# Display
# =====================================================================

echo
echo "============================================================"
echo " Illumio PCE 26.2.10 Dependency Downloader"
echo "============================================================"
echo
echo "Rocky Linux:"
echo "  ${VERSION}"
echo
echo "Architecture:"
echo "  ${ARCH}"
echo
echo "Output:"
echo "  ${OUTPUT_DIR}"
echo

echo "Repositories:"
echo
echo "BaseOS:"
echo "  ${BASEOS_URL}"
echo
echo "AppStream:"
echo "  ${APPSTREAM_URL}"
echo
echo "CRB:"
echo "  ${CRB_URL}"
echo


# =====================================================================
# Display Package List
# =====================================================================

echo
echo "============================================================"
echo " PCE Required Packages"
echo "============================================================"
echo


for pkg in "${PACKAGES[@]}"; do

    echo "  ${pkg}"

done


# =====================================================================
# Clean Old RPM
# =====================================================================

info "清除 Rocky ${VERSION} 舊 RPM"


rm -f "${RPM_DIR}"/*.rpm


# =====================================================================
# Download
# =====================================================================
#
# --resolve
#
#   解析這些 PCE prerequisite 自己需要的 dependencies。
#
#
# --alldeps
#
#   即使 dependency 已經安裝在下載主機，
#   仍然下載完整 RPM。
#
#
# --disablerepo="*"
#
#   完全不用目前主機自己的 repository。
#
#
# 所以即使你是在：
#
#   Rocky 10.2
#
# 上執行：
#
#   ./pce-deps.sh 9.6
#
# 仍然只會下載 Rocky 9.6 RPM。
#
# =====================================================================


info "下載 Rocky ${VERSION} PCE prerequisites + dependencies"


DOWNLOAD_LOG="${OUTPUT_DIR}/download.log"


set +e


dnf download \
    --resolve \
    --alldeps \
    --archlist="${ARCH},noarch" \
    --destdir="${RPM_DIR}" \
    --disablerepo="*" \
    --repofrompath="pce-${VERSION}-baseos,${BASEOS_URL}" \
    --repofrompath="pce-${VERSION}-appstream,${APPSTREAM_URL}" \
    --repofrompath="pce-${VERSION}-crb,${CRB_URL}" \
    --enablerepo="pce-${VERSION}-baseos" \
    --enablerepo="pce-${VERSION}-appstream" \
    --enablerepo="pce-${VERSION}-crb" \
    "${PACKAGES[@]}" \
    2>&1 | tee "${DOWNLOAD_LOG}"


RESULT=${PIPESTATUS[0]}


set -e


# =====================================================================
# Download Failure
# =====================================================================

if [[ "${RESULT}" -ne 0 ]]; then

    echo
    echo "============================================================"
    echo " Rocky Linux ${VERSION} : FAILED"
    echo "============================================================"
    echo
    echo "DNF resolver 發生錯誤。"
    echo
    echo "錯誤摘要："
    echo


    grep -Ei \
        "No match|No package|nothing provides|conflicting|cannot install|failed|error:" \
        "${DOWNLOAD_LOG}" \
        || true


    echo
    echo "完整 Log："
    echo
    echo "  ${DOWNLOAD_LOG}"
    echo

    exit 1

fi


# =====================================================================
# RPM Count
# =====================================================================

RPM_COUNT=$(find "${RPM_DIR}" \
    -maxdepth 1 \
    -type f \
    -name '*.rpm' \
    | wc -l)


if [[ "${RPM_COUNT}" -eq 0 ]]; then

    die "沒有下載到 RPM"

fi


info "下載完成：${RPM_COUNT} 個 RPM"


# =====================================================================
# Verify Main Packages
# =====================================================================
#
# 確認主要 package 本身都有在 bundle 裡。
#
# =====================================================================

VERIFY_FILE="${OUTPUT_DIR}/verification.txt"


: > "${VERIFY_FILE}"


MISSING=0


info "驗證 PCE prerequisite RPM"


for pkg in "${PACKAGES[@]}"; do

    FOUND=0


    while IFS= read -r rpmfile; do

        RPM_NAME=$(rpm -qp \
            --queryformat '%{NAME}' \
            "${rpmfile}" \
            2>/dev/null || true)


        if [[ "${RPM_NAME}" == "${pkg}" ]]; then

            FOUND=1
            break

        fi

    done < <(

        find "${RPM_DIR}" \
            -maxdepth 1 \
            -type f \
            -name '*.rpm'

    )


    if [[ "${FOUND}" -eq 1 ]]; then

        printf "%-25s OK\n" "${pkg}" | tee -a "${VERIFY_FILE}"

    else

        printf "%-25s MISSING\n" "${pkg}" | tee -a "${VERIFY_FILE}"

        MISSING=$((MISSING + 1))

    fi

done


if [[ "${MISSING}" -ne 0 ]]; then

    warn "${MISSING} 個主要 package 沒有出現在 bundle"

fi


# =====================================================================
# Manifest
# =====================================================================

MANIFEST="${OUTPUT_DIR}/manifest.txt"


info "建立 manifest.txt"


{
    echo "============================================================"
    echo "Illumio PCE 26.2.10 Offline Dependency Bundle"
    echo "============================================================"
    echo
    echo "Rocky Linux:"
    echo "  ${VERSION}"
    echo
    echo "Architecture:"
    echo "  ${ARCH}"
    echo
    echo "Generated:"
    echo "  $(date)"
    echo


    echo
    echo "============================================================"
    echo "Requested PCE Packages"
    echo "============================================================"
    echo


    for pkg in "${PACKAGES[@]}"; do

        echo "${pkg}"

    done


    echo
    echo "============================================================"
    echo "Downloaded RPM"
    echo "============================================================"
    echo


    while IFS= read -r rpmfile; do

        rpm -qp \
            --queryformat '%{NAME}-%{VERSION}-%{RELEASE}.%{ARCH}\n' \
            "${rpmfile}" \
            2>/dev/null || basename "${rpmfile}"

    done < <(

        find "${RPM_DIR}" \
            -maxdepth 1 \
            -type f \
            -name '*.rpm' \
            | sort

    )


} > "${MANIFEST}"


# =====================================================================
# packages.txt
# =====================================================================

PACKAGE_LIST="${OUTPUT_DIR}/packages.txt"


info "建立 packages.txt"


find "${RPM_DIR}" \
    -maxdepth 1 \
    -type f \
    -name '*.rpm' \
    -printf '%f\n' \
    | sort \
    > "${PACKAGE_LIST}"


# =====================================================================
# SHA256
# =====================================================================

SHA_FILE="${OUTPUT_DIR}/SHA256SUMS"


info "建立 SHA256SUMS"


(
    cd "${RPM_DIR}"

    sha256sum ./*.rpm > "${SHA_FILE}"

)


# =====================================================================
# Result
# =====================================================================

echo
echo
echo "============================================================"
echo " SUCCESS"
echo "============================================================"
echo
echo "Illumio:"
echo "  PCE 26.2.10"
echo
echo "Rocky Linux:"
echo "  ${VERSION}"
echo
echo "Requested packages:"
echo "  ${#PACKAGES[@]}"
echo
echo "Downloaded RPM:"
echo "  ${RPM_COUNT}"
echo
echo "RPM Directory:"
echo "  ${RPM_DIR}"
echo
echo "Verification:"
echo "  ${VERIFY_FILE}"
echo
echo "Manifest:"
echo "  ${MANIFEST}"
echo
echo "Package list:"
echo "  ${PACKAGE_LIST}"
echo
echo "Checksum:"
echo "  ${SHA_FILE}"
echo
echo "============================================================"

#!/bin/bash

# =====================================================================
# Rocky Linux 9.x / 10.x Offline Dependency Downloader
#
# 支援版本：
#
#   Rocky Linux 9.0 ~ 9.8
#   Rocky Linux 10.0 ~ 10.2
#
#
# 功能：
#
#   1. 一次指定一個 Rocky Linux 版本
#   2. 下載指定的套件
#   3. 使用 --resolve 補齊 dependencies
#   4. 使用 --alldeps，即使 dependency 已安裝在下載主機，
#      仍然下載 RPM
#   5. 完全停用目前下載主機自己的 repository
#   6. 使用指定 Rocky Linux 版本的官方 repository
#   7. 不打包 tar.gz
#   8. 不加入 Illumio VEN RPM
#
#
# 使用方式：
#
#   ./rocky-deps.sh 9.6
#
#   ./rocky-deps.sh 10.2
#
# =====================================================================

set -Eeuo pipefail


# =====================================================================
# 要下載的套件
#
# 這裡包含：
#
#   - Illumio 官方要求的套件
#   - 你另外指定的系統套件
#
# =====================================================================

PACKAGES=(

    # ================================================================
    # 你額外指定的套件
    # ================================================================

    bash
    bzip2
    coreutils
    findutils
    procps-ng

    libxml2
    zlib
    xz-libs
    libuuid
    ncurses-libs
    openssl-libs
    libxcrypt
    libstdc++
    libgcc
    glibc


    # ================================================================
    # Illumio 官方要求套件
    # ================================================================

    bind-utils
    diffutils
    curl
    gawk
    gmp
    gzip
    libcap
    libnfnetlink
    net-tools
    nftables
    sed
    shadow-utils
    tar
    util-linux

)


# =====================================================================
# 基本設定
# =====================================================================

ARCH="x86_64"

OUTPUT_ROOT="/agent-download/rocky-offline-dependencies"


# =====================================================================
# Functions
# =====================================================================

info()
{
    echo
    echo "[INFO] $*"
}


warn()
{
    echo
    echo "[WARN] $*"
}


die()
{
    echo
    echo "[ERROR] $*" >&2
    exit 1
}


# =====================================================================
# Root check
# =====================================================================

if [[ "${EUID}" -ne 0 ]]; then

    die "請使用 root 執行此腳本"

fi


# =====================================================================
# 參數檢查
# =====================================================================

if [[ $# -ne 1 ]]; then

    echo
    echo "使用方式："
    echo
    echo "  $0 <Rocky版本>"
    echo
    echo "例如："
    echo
    echo "  $0 9.6"
    echo "  $0 10.2"
    echo

    exit 1

fi


VERSION="$1"


# =====================================================================
# 允許的 Rocky Linux 版本
# =====================================================================

case "${VERSION}" in

    9.0|9.1|9.2|9.3|9.4|9.5|9.6|9.7|9.8|\
    10.0|10.1|10.2)

        ;;

    *)

        die "只支援 Rocky Linux 9.0 ~ 9.8、10.0 ~ 10.2"

        ;;

esac


# =====================================================================
# 取得 Major Version
#
# 例如：
#
#   9.6  -> 9
#   10.2 -> 10
#
# =====================================================================

MAJOR="${VERSION%%.*}"


# =====================================================================
# 檢查 dnf
# =====================================================================

if ! command -v dnf >/dev/null 2>&1; then

    die "找不到 dnf"

fi


# =====================================================================
# 檢查 dnf download
# =====================================================================

if ! dnf download --help >/dev/null 2>&1; then

    info "找不到 dnf download，嘗試安裝 dnf-plugins-core"

    dnf install -y dnf-plugins-core || true

fi


if ! dnf download --help >/dev/null 2>&1; then

    die "dnf download 無法使用"

fi


# =====================================================================
# Output directory
# =====================================================================

OUTPUT_DIR="${OUTPUT_ROOT}/rocky-${VERSION}"

RPM_DIR="${OUTPUT_DIR}/packages"


mkdir -p "${RPM_DIR}"


# =====================================================================
# 判斷 Rocky Repository
# =====================================================================
#
# 目前設定：
#
# Rocky 9.0 ~ 9.8
#   全部固定使用 Vault
#
# Rocky 10.0
#   Vault
#
# Rocky 10.1
#   Vault
#
# Rocky 10.2
#   Current repository
#
#
# 這樣有一個好處：
#
#   不會因為未來 Rocky minor version 更新，
#   導致你抓到不同版本的 RPM。
#
# =====================================================================


case "${VERSION}" in

    10.2)

        # -------------------------------------------------------------
        # Rocky Linux 10.2
        #
        # 目前是正式 repository
        # -------------------------------------------------------------

        BASE_URL="https://dl.rockylinux.org/pub/rocky/${VERSION}"
        ;;


    *)

        # -------------------------------------------------------------
        # Rocky 9.0 ~ 9.8
        # Rocky 10.0 ~ 10.1
        #
        # 固定使用 Vault
        # -------------------------------------------------------------

        BASE_URL="https://dl.rockylinux.org/vault/rocky/${VERSION}"
        ;;

esac


# =====================================================================
# Repository URLs
# =====================================================================

BASEOS_URL="${BASE_URL}/BaseOS/${ARCH}/os/"

APPSTREAM_URL="${BASE_URL}/AppStream/${ARCH}/os/"

CRB_URL="${BASE_URL}/CRB/${ARCH}/os/"


# =====================================================================
# 顯示設定
# =====================================================================

echo
echo "============================================================"
echo " Rocky Linux Offline Dependency Downloader"
echo "============================================================"
echo
echo "Rocky Version : ${VERSION}"
echo "Major Version : ${MAJOR}"
echo "Architecture  : ${ARCH}"
echo
echo "Output:"
echo
echo "  ${OUTPUT_DIR}"
echo
echo "Repository:"
echo
echo "  BaseOS:"
echo "  ${BASEOS_URL}"
echo
echo "  AppStream:"
echo "  ${APPSTREAM_URL}"
echo
echo "  CRB:"
echo "  ${CRB_URL}"
echo


# =====================================================================
# 顯示套件清單
# =====================================================================

echo
echo "============================================================"
echo " Requested Packages"
echo "============================================================"
echo


for pkg in "${PACKAGES[@]}"; do

    echo "  ${pkg}"

done


# =====================================================================
# 清除之前下載的 RPM
# =====================================================================

info "清除 Rocky ${VERSION} 舊 RPM"


rm -f "${RPM_DIR}"/*.rpm


# =====================================================================
# Download
# =====================================================================
#
# --resolve
#
#   自動解析 PACKAGES 所需要的 dependency
#
#
# --alldeps
#
#   即使 dependency 已經裝在目前這台下載主機，
#   還是下載。
#
#
# --archlist
#
#   只允許：
#
#      x86_64
#      noarch
#
#
# --disablerepo="*"
#
#   不使用下載主機自己的 repository。
#
#
# --repofrompath
#
#   直接指定我們要的 Rocky Linux repository。
#
# =====================================================================


info "下載 Rocky ${VERSION} packages + all dependencies"


DOWNLOAD_LOG="${OUTPUT_DIR}/download.log"


set +e


dnf download \
    --resolve \
    --alldeps \
    --archlist="${ARCH},noarch" \
    --destdir="${RPM_DIR}" \
    --disablerepo="*" \
    --repofrompath="rocky-${VERSION}-baseos,${BASEOS_URL}" \
    --repofrompath="rocky-${VERSION}-appstream,${APPSTREAM_URL}" \
    --repofrompath="rocky-${VERSION}-crb,${CRB_URL}" \
    --enablerepo="rocky-${VERSION}-baseos" \
    --enablerepo="rocky-${VERSION}-appstream" \
    --enablerepo="rocky-${VERSION}-crb" \
    "${PACKAGES[@]}" \
    2>&1 | tee "${DOWNLOAD_LOG}"


RESULT=${PIPESTATUS[0]}


set -e


# =====================================================================
# Download error
# =====================================================================

if [[ "${RESULT}" -ne 0 ]]; then

    echo
    echo "============================================================"
    echo " Rocky Linux ${VERSION} : FAILED"
    echo "============================================================"
    echo
    echo "DNF dependency resolver/download 發生錯誤。"
    echo
    echo "錯誤摘要："
    echo


    grep -Ei \
        "No match|No package|nothing provides|conflicting|cannot install|failed|error:" \
        "${DOWNLOAD_LOG}" \
        || true


    echo
    echo "完整 Log："
    echo
    echo "  ${DOWNLOAD_LOG}"
    echo

    exit 1

fi


# =====================================================================
# RPM Count
# =====================================================================

RPM_COUNT=$(find "${RPM_DIR}" \
    -maxdepth 1 \
    -type f \
    -name '*.rpm' \
    | wc -l)


if [[ "${RPM_COUNT}" -eq 0 ]]; then

    die "沒有下載到任何 RPM"

fi


info "共下載 ${RPM_COUNT} 個 RPM"


# =====================================================================
# 建立 package manifest
# =====================================================================

MANIFEST="${OUTPUT_DIR}/manifest.txt"


info "建立 manifest.txt"


{
    echo "============================================================"
    echo "Rocky Linux Offline Dependency Bundle"
    echo "============================================================"
    echo
    echo "Rocky Linux:"
    echo "  ${VERSION}"
    echo
    echo "Architecture:"
    echo "  ${ARCH}"
    echo
    echo "Generated:"
    echo "  $(date)"
    echo


    echo
    echo "============================================================"
    echo "Requested Packages"
    echo "============================================================"
    echo


    for pkg in "${PACKAGES[@]}"; do

        echo "${pkg}"

    done


    echo
    echo "============================================================"
    echo "Downloaded RPM"
    echo "============================================================"
    echo


    while IFS= read -r rpmfile; do

        rpm -qp \
            --queryformat '%{NAME}-%{VERSION}-%{RELEASE}.%{ARCH}\n' \
            "${rpmfile}" \
            2>/dev/null || basename "${rpmfile}"

    done < <(

        find "${RPM_DIR}" \
            -maxdepth 1 \
            -type f \
            -name '*.rpm' \
            | sort

    )


} > "${MANIFEST}"


# =====================================================================
# 建立純檔名列表
# =====================================================================

PACKAGE_LIST="${OUTPUT_DIR}/packages.txt"


info "建立 packages.txt"


find "${RPM_DIR}" \
    -maxdepth 1 \
    -type f \
    -name '*.rpm' \
    -printf '%f\n' \
    | sort \
    > "${PACKAGE_LIST}"


# =====================================================================
# 建立 SHA256
# =====================================================================

SHA_FILE="${OUTPUT_DIR}/SHA256SUMS"


info "建立 SHA256SUMS"


(
    cd "${RPM_DIR}"

    sha256sum ./*.rpm > "${SHA_FILE}"

)


# =====================================================================
# 完成
# =====================================================================

echo
echo
echo "============================================================"
echo " SUCCESS"
echo "============================================================"
echo
echo "Rocky Linux:"
echo "  ${VERSION}"
echo
echo "Requested package count:"
echo "  ${#PACKAGES[@]}"
echo
echo "Downloaded RPM:"
echo "  ${RPM_COUNT}"
echo
echo "RPM Directory:"
echo "  ${RPM_DIR}"
echo
echo "Manifest:"
echo "  ${MANIFEST}"
echo
echo "Package List:"
echo "  ${PACKAGE_LIST}"
echo
echo "Checksum:"
echo "  ${SHA_FILE}"
echo
echo "============================================================"
